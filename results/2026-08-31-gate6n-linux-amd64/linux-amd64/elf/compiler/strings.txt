@child

	jnz %signaled, @signal, @exited
@signal
	%sr =w add %sig, 128

@exited

	%code =w and %shifted, 255
	ret %code
@fail
	ret 127

function l $g4_gc_alloc(l %descriptor, l %payload_size, l %reserved_size) {

	storel %new_allocated, $g4_gc_allocated_bytes

function $g4_gc_mark(l %object) {
@start
	%nil =w ceql %object, 0
	jnz %nil, @finish, @descriptor
@descriptor
	%descriptor_mark =l loadl %object
	%static =w ceql %descriptor_mark, 0
	jnz %static, @finish, @check_mark
@check_mark
	%mark =l and %descriptor_mark, 1
	%seen =w cnel %mark, 0
	jnz %seen, @finish, @dispatch
@dispatch
	%marked_descriptor =l or %descriptor_mark, 1
	storel %marked_descriptor, %object
	%descriptor =l and %descriptor_mark, -2
	%kind =l loadl %descriptor
	%fixed =w ceql %kind, 1
	jnz %fixed, @fixed, @check_array
@fixed
	call $g4_gc_scan_fixed(l %object, l %descriptor)
	ret
@check_array
	%array =w ceql %kind, 2
	jnz %array, @array, @finish
@array
	call $g4_gc_scan_array(l %object, l %descriptor)
@finish
	ret
function $g4_gc_mark_roots() {
@start
	%temp_count =l loadl $g4_gc_temp_count
	%temp_data =l loadl $g4_gc_temp_data
	jmp @temp_loop
@temp_loop

	jmp @temp_loop
@globals
	jmp @global_loop
@global_loop

@finish
	ret
function $g4_gc_sweep() {
@start
	%head =l loadl $g4_gc_heap
	jmp @loop
@loop
	%current =l phi @start %head, @advance %next
	%previous =l phi @start 0, @advance %next_previous
	%done =w ceql %current, 0
	jnz %done, @finish, @object
@object
	%next =l loadl %current
	%descriptor_address =l add %current, 8
	%descriptor_mark =l loadl %descriptor_address
	%mark =l and %descriptor_mark, 1
	%live =w cnel %mark, 0
	jnz %live, @keep, @release
@keep
	%descriptor =l and %descriptor_mark, -2
	storel %descriptor, %descriptor_address
	jmp @advance
@release
	%released_descriptor =l and %descriptor_mark, -2
	%kind =l loadl %released_descriptor
	%array =w ceql %kind, 2
	jnz %array, @release_array, @check_fixed
@release_array
	%data_address =l add %current, 32
	%data =l loadl %data_address
	%capacity_address =l add %current, 24
	%capacity =l loadl %capacity_address
	%backing_bytes =l mul %capacity, 8
	%heap_with_backing =l loadl $g4_gc_heap_bytes
	%heap_without_backing =l sub %heap_with_backing, %backing_bytes
	storel %heap_without_backing, $g4_gc_heap_bytes
	%reclaimed_before_backing =l loadl $g4_gc_reclaimed_bytes
	%reclaimed_with_backing =l add %reclaimed_before_backing, %backing_bytes
	storel %reclaimed_with_backing, $g4_gc_reclaimed_bytes
	%has_data =w cnel %data, 0
	jnz %has_data, @free_data, @array_ready
@free_data
	call $free(l %data)
	jmp @array_ready
@array_ready
	%array_size =l copy 40
	jmp @unlink
@check_fixed
	%fixed =w ceql %kind, 1
	jnz %fixed, @fixed_size, @string_size
@fixed_size
	%payload_size_address =l add %released_descriptor, 16
	%payload_size =l loadl %payload_size_address
	%fixed_size =l add %payload_size, 16
	jmp @unlink
@string_size
	%length_address =l add %current, 16
	%length =l loadl %length_address
	%string_size =l add %length, 25
	jmp @unlink
@unlink
	%size =l phi @array_ready %array_size, @fixed_size %fixed_size, @string_size %string_size
	%at_head =w ceql %previous, 0
	jnz %at_head, @unlink_head, @unlink_after
@unlink_head
	storel %next, $g4_gc_heap
	jmp @account
@unlink_after
	storel %next, %previous
@account
	%bytes =l loadl $g4_gc_heap_bytes
	%remaining =l sub %bytes, %size
	storel %remaining, $g4_gc_heap_bytes
	%reclaimed =l loadl $g4_gc_reclaimed_bytes
	%new_reclaimed =l add %reclaimed, %size
	storel %new_reclaimed, $g4_gc_reclaimed_bytes
	call $free(l %current)
	jmp @advance
@advance
	%next_previous =l phi @keep %current, @account %previous
	jmp @loop
@finish
	ret

@start

	%count =l loadl $g4_gc_collection_count

	storel %next_automatic, $g4_gc_automatic_collection_count
@pace
	%live =l loadl $g4_gc_heap_bytes
	%doubled =l mul %live, 2
	%candidate =l add %doubled, 65536
	%small =w csltl %candidate, 1048576
	jnz %small, @minimum, @candidate

function l $g4_string_to_integer(l %string) {

@unsigned

@validate_next
	%validation_next =l add %validation_index, 1
	jmp @validate
@parse_start
	%parse_begin =l loadl %parse_start_slot

@parse

@parse_next
